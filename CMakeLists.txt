cmake_minimum_required(VERSION 4.4.5)
project(ts808 VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_VERSION VERSION_EQUAL "4.4.5")
  message(FATAL_ERROR "CMake 4.4.5 required")
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "15.3.0")
  message(FATAL_ERROR "Qualification requires GCC 15.3.0")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(ts808_json INTERFACE)
target_include_directories(ts808_json INTERFACE "${CMAKE_SOURCE_DIR}/.tools/deps/json/single_include")
option(TS808_CHECKED "Keep assertions and Eigen allocation checks" OFF)
# Dependencies are staged by setup; configuring and running never download.
set(EIGEN_ROOT "${CMAKE_SOURCE_DIR}/.tools/deps/eigen-5.0.1" CACHE PATH "Eigen 5.0.1 source")
set(CATCH_ROOT "${CMAKE_SOURCE_DIR}/.tools/deps/Catch2-3.16.1" CACHE PATH "Catch2 3.16.1 source")
set(JSON_ROOT "${CMAKE_SOURCE_DIR}/.tools/deps/json" CACHE PATH "nlohmann/json 3.12.0 source")
file(READ "${EIGEN_ROOT}/Eigen/Version" EIGEN_VERSION_TEXT)
if(NOT EIGEN_VERSION_TEXT MATCHES "5.0.1")
  message(FATAL_ERROR "Eigen 5.0.1 is required")
endif()
add_library(ts808 STATIC src/devices.cpp src/stages.cpp src/block_solve.cpp src/newton.cpp src/model.cpp src/linear_response.cpp)
target_include_directories(ts808 PUBLIC include)
target_include_directories(ts808 SYSTEM PUBLIC "${EIGEN_ROOT}")
target_compile_options(ts808 PUBLIC -O2 -fno-fast-math -ffp-contract=off)
target_compile_options(ts808 PRIVATE -Wall -Wextra -Wpedantic)
if(TS808_CHECKED)
  target_compile_definitions(ts808 PUBLIC EIGEN_RUNTIME_NO_MALLOC)
  target_compile_options(ts808 PUBLIC -UNDEBUG)
endif()
if(BUILD_TESTING)
  set(CATCH_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CATCH_ROOT}" "${CMAKE_BINARY_DIR}/catch2" EXCLUDE_FROM_ALL)
  if(NOT Catch2_VERSION VERSION_EQUAL "3.16.1")
    # Catch2_VERSION is scoped in upstream's subdirectory; inspect the source identity.
    file(STRINGS "${CATCH_ROOT}/CMakeLists.txt" CATCH_PROJECT REGEX "VERSION 3.16.1")
    if(NOT CATCH_PROJECT)
      message(FATAL_ERROR "Catch2 3.16.1 is required")
    endif()
  endif()
  add_executable(ts808_tests tests/test_main.cpp tests/device_jacobians.cpp tests/control_endpoints.cpp tests/coupled_solver.cpp tests/linear_and_history.cpp tests/allocation_guard.cpp tests/linear_response.cpp tests/conditioning_and_scaling.cpp)
  target_link_libraries(ts808_tests PRIVATE ts808 Catch2::Catch2)
  add_test(NAME core COMMAND ts808_tests)
endif()

add_executable(ts808_offline app/offline_main.cpp app/run_io.cpp)
target_link_libraries(ts808_offline PRIVATE ts808 ts808_json)
